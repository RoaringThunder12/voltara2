# .0001 -- The Masks Must Meet. A player asked to call the Conclave: the planner opens from here.
spn_conclave.0001 = {
	type = character_event
	title = spn_conclave.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:spn_conclave_reason ?= flag:chair }
				desc = spn_conclave.0001.desc_chair
			}
			triggered_desc = {
				trigger = { var:spn_conclave_reason ?= flag:prerogative }
				desc = spn_conclave.0001.desc_prerogative
			}
			desc = spn_conclave.0001.desc_will
		}
		triggered_desc = {
			trigger = { exists = scope:spn_cv_second }
			desc = spn_conclave.0001.desc_second
		}
	}
	theme = learning
	override_background = { reference = magic_temple }
	left_portrait = { character = root animation = personality_rational }
	# 2.38c: your second, the highest member of your temple after you (Thomas: "so it doesn't feel like it's just me").
	right_portrait = {
		character = scope:spn_cv_second
		trigger = { exists = scope:spn_cv_second }
		animation = personality_bold
	}
	trigger = {
		is_alive = yes
		has_character_flag = spn_conclave_may_call
		NOT = { involved_activity ?= { has_activity_type = spn_conclave } } # 2.38c: already planning or travelling to it
	}
	immediate = {
		# The named heir first, as the succession does (spn_order_succession_effect); else the highest member.
		var:spn_order_chapter.var:spn_order_heir ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { this = root }
				}
				save_scope_as = spn_cv_second
			}
		}
		var:spn_order_chapter ?= {
			ordered_in_list = {
				variable = spn_order_members
				limit = {
					is_alive = yes
					NOT = { this = root }
					spn_is_order_member_trigger = yes
				}
				order_by = spn_order_succession_score_value
				save_scope_as = spn_cv_second_alt
			}
		}
		if = {
			limit = { NOT = { exists = scope:spn_cv_second } }
			scope:spn_cv_second_alt ?= { save_scope_as = spn_cv_second }
		}
	}
	option = {
		name = spn_conclave.0001.a
		open_view_data = {
			view = activity_list_detail_host_window
			data = activity_type:spn_conclave
		}
	}
	option = {
		name = spn_conclave.0001.b
		custom_tooltip = spn_conclave.0001.b.tt
	}
}

# THE MASKS: the weekly pool. The host's pulse hands one moment to a player who came and has not had it.
# ==================================================================================================
spn_conclave.0099 = {
	type = activity_event
	hidden = yes
	immediate = {
		scope:activity = {
			random_attending_character = {
				limit = {
					is_ai = no
					NOT = { has_character_flag = spn_cv_moment_seen }
				}
				add_character_flag = { flag = spn_cv_moment_seen days = 60 }
				random_list = {
					10 = {
						trigger = { NOT = { has_character_flag = spn_cv_seen_familiar } }
						trigger_event = spn_conclave.0100
					}
					10 = {
						trigger = { NOT = { has_character_flag = spn_cv_seen_story } }
						trigger_event = spn_conclave.0110
					}
					10 = {
						trigger = { NOT = { has_character_flag = spn_cv_seen_knows } }
						trigger_event = spn_conclave.0120
					}
				}
			}
		}
	}
}

# .0100 -- A Familiar Mask. Someone from another temple.
spn_conclave.0100 = {
	type = activity_event
	title = spn_conclave.0100.t
	desc = spn_conclave.0100.desc
	theme = learning
	override_background = { reference = magic_temple }
	left_portrait = { character = root animation = interested }
	right_portrait = { character = scope:spn_cv_stranger animation = personality_rational }
	trigger = {
		is_alive = yes
		scope:activity = {
			any_attending_character = {
				spn_is_order_member_trigger = yes
				NOT = { this = root }
				NOT = { var:spn_order_chapter ?= root.var:spn_order_chapter }
			}
		}
	}
	immediate = {
		add_character_flag = { flag = spn_cv_seen_familiar years = 20 }
		scope:activity = {
			random_attending_character = {
				limit = {
					spn_is_order_member_trigger = yes
					NOT = { this = root }
					NOT = { var:spn_order_chapter ?= root.var:spn_order_chapter }
				}
				save_scope_as = spn_cv_stranger
			}
		}
	}
	option = {
		name = spn_conclave.0100.a
		custom_tooltip = spn_conclave.0100.a.tt
		spn_order_reveal_membership_effect = { WHO = scope:spn_cv_stranger TO = root }
		spn_order_reveal_membership_effect = { WHO = root TO = scope:spn_cv_stranger }
		add_opinion = { target = scope:spn_cv_stranger modifier = friendliness_opinion opinion = 15 }
		reverse_add_opinion = { target = scope:spn_cv_stranger modifier = friendliness_opinion opinion = 15 }
	}
	option = {
		name = spn_conclave.0100.b
		custom_tooltip = spn_conclave.0100.b.tt
		spn_order_add_standing_effect = { VALUE = 3 }
	}
}

# .0410 -- Fill the Seats. A player Grand Magus names one member to an empty seat; again while seats and names remain
# (at most three times: spn_cv_seat_rounds).
spn_conclave.0410 = {
	type = character_event
	title = spn_conclave.0410.t
	desc = spn_conclave.0410.desc
	theme = learning
	override_background = { reference = magic_temple }
	left_portrait = { character = root animation = personality_rational }
	trigger = {
		is_alive = yes
		global_var:current_archmage_global ?= root
		spn_conclave_seat_open_trigger = yes
	}
	immediate = {
		save_scope_as = spn_cv_chair
		if = {
			limit = { NOT = { exists = var:spn_cv_seat_rounds } }
			set_variable = { name = spn_cv_seat_rounds value = 0 days = 30 }
		}
		change_variable = { name = spn_cv_seat_rounds add = 1 }
		every_in_global_list = {
			variable = spn_order_chapters
			limit = { exists = var:spn_order_chapter }
			var:spn_order_chapter = {
				every_in_list = {
					variable = spn_order_members
					limit = { spn_order_seat_candidate_trigger = { CHAIR = scope:spn_cv_chair } }
					add_to_temporary_list = spn_cv_nominees
				}
			}
		}
		ordered_in_list = {
			list = spn_cv_nominees
			order_by = spn_conclave_seat_score_value
			position = 0
			check_range_bounds = no
			save_scope_as = spn_cv_n1
		}
		ordered_in_list = {
			list = spn_cv_nominees
			order_by = spn_conclave_seat_score_value
			position = 1
			check_range_bounds = no
			save_scope_as = spn_cv_n2
		}
		ordered_in_list = {
			list = spn_cv_nominees
			order_by = spn_conclave_seat_score_value
			position = 2
			check_range_bounds = no
			save_scope_as = spn_cv_n3
		}
	}
	option = {
		name = spn_conclave.0410.n1
		trigger = { exists = scope:spn_cv_n1 }
		scope:spn_cv_n1 = { spn_order_take_seat_effect = yes }
		spn_conclave_seat_again_effect = yes
	}
	option = {
		name = spn_conclave.0410.n2
		trigger = { exists = scope:spn_cv_n2 }
		scope:spn_cv_n2 = { spn_order_take_seat_effect = yes }
		spn_conclave_seat_again_effect = yes
	}
	option = {
		name = spn_conclave.0410.n3
		trigger = { exists = scope:spn_cv_n3 }
		scope:spn_cv_n3 = { spn_order_take_seat_effect = yes }
		spn_conclave_seat_again_effect = yes
	}
	option = {
		name = spn_conclave.0410.none
		custom_tooltip = spn_conclave.0410.none.tt
	}
}
