# Call to war as attacker
great_empire_war.0005 = {
	type = character_event

    hidden = yes

	immediate = {
		if = {
			limit = { 
				NOT = {
				    primary_title.tier = tier_barony
			    }
			}
			scope:war = { add_attacker = ROOT }
			add_character_modifier = {
				modifier = great_empire_war_modifier
				days = 3650
			}
		}
	}
}

# Tribalize defeated empire
great_empire_war.0007 = {
	type = character_event

    hidden = yes

	immediate = {
		if = {
			limit = { 
                AND = {			
					has_game_rule = shattered_tribalism									
			        NOT = {
				        has_government = republic_government
			        }
			        NOT = {
				        has_government = theocracy_government
			        }
		        }
			}
			change_government = tribal_government
			# add_prestige = -4000
			# add_piety = -4000
		}
	}
}
