namespace = interactive
interactive.0013 = {
	hidden = yes
	immediate = {
		if = {
			limit = {
				exists = var:miv_defect_enemy_spokesman
				exists = var:miv_defect_enemy_batchwar
				exists = var:miv_defect_enemy_batchside
			}
			var:miv_defect_enemy_batchwar = {
				save_scope_as = miv_batch_war
			}
			var:miv_defect_enemy_batchside = {
				save_scope_as = miv_batch_side
			}
			var:miv_defect_enemy_spokesman = {
				save_scope_as = miv_flush_vassal
			}
			if = {
				limit = {
					scope:miv_flush_vassal = {
						exists = var:miv_defect_enemy_fellows
						var:miv_defect_enemy_fellows >= 1
					}
				}
				save_temporary_scope_value_as = {
					name = miv_toast_grouped
					value = yes
				}
				interactive_bannermen_toast_side_viewer_effect = {
					WAR = scope:miv_batch_war
					SIDE_OF = scope:miv_batch_side
					VASSAL = scope:miv_flush_vassal
					TYPE = msg_war_oathbreaker
					TITLE = miv_toast_defect_enemy_many
					TOOLTIP = miv_toast_defect_enemy_many_desc
				}
				scope:miv_flush_vassal = {
					remove_variable = miv_defect_enemy_fellows
				}
			}
			else = {
				interactive_bannermen_toast_side_viewer_effect = {
					WAR = scope:miv_batch_war
					SIDE_OF = scope:miv_batch_side
					VASSAL = scope:miv_flush_vassal
					TYPE = msg_war_oathbreaker
					TITLE = miv_toast_defect_enemy_one
					TOOLTIP = miv_toast_defect_enemy_one_desc
				}
			}
		}
		remove_variable = miv_defect_enemy_spokesman
		remove_variable = miv_defect_enemy_pending
		remove_variable = miv_defect_enemy_batchwar
		remove_variable = miv_defect_enemy_batchside
		if = {
			limit = {
				has_variable_list = miv_defect_enemy_overflow
			}
			every_in_list = {
				variable = miv_defect_enemy_overflow
				add_to_temporary_list = miv_defect_enemy_replay
			}
			clear_variable_list = miv_defect_enemy_overflow
			every_in_list = {
				list = miv_defect_enemy_replay
				save_temporary_scope_as = miv_replay_vassal
				if = {
					limit = {
						is_alive = yes
						exists = var:miv_defect_enemy_overflow_war
						exists = var:miv_defect_enemy_overflow_side
					}
					var:miv_defect_enemy_overflow_war = {
						save_temporary_scope_as = miv_replay_war
					}
					var:miv_defect_enemy_overflow_side = {
						save_temporary_scope_as = miv_replay_side
					}
					remove_variable = miv_defect_enemy_overflow_war
					remove_variable = miv_defect_enemy_overflow_side
					root = {
						interactive_bannermen_batch_toast_effect = {
							KEY = defect_enemy
							EVENT = 0013
							HOLDER = root
							WAR = scope:miv_replay_war
							SIDE_OF = scope:miv_replay_side
							VASSAL = scope:miv_replay_vassal
						}
					}
				}
				else = {
					remove_variable = miv_defect_enemy_overflow_war
					remove_variable = miv_defect_enemy_overflow_side
				}
			}
		}
	}
}

interactive.0001 = {
	title = interactive.0001.title
	desc = interactive.0001.desc
	theme = war
	left_portrait = scope:defender
	right_portrait = scope:attacker
	option = {
		name = interactive.0001.a
		save_scope_as = recipient
		if = {
			limit = {
				is_vassal_or_below_of = scope:attacker
			}
			scope:war = {
				hidden_effect = {
					set_called_to = scope:recipient
				}
				add_attacker = prev
			}
			scope:attacker = {
				add_opinion = {
					modifier = grateful_opinion
					target = scope:recipient
					opinion = 10
				}
			}
		}
		else_if = {
			limit = {
				is_vassal_or_below_of = scope:defender
			}
			scope:war = {
				hidden_effect = {
					set_called_to = scope:recipient
				}
				add_defender = prev
			}
			scope:defender = {
				add_opinion = {
					modifier = grateful_opinion
					target = scope:recipient
					opinion = 10
				}
			}
		}
		if = {
			limit = {
				government_has_flag = government_has_influence
				scope:recipient.liege ?= {
					government_has_flag = government_has_influence
				}
				OR = {
					is_vassal_or_below_of = scope:attacker
					is_vassal_or_below_of = scope:defender
				}
			}
			change_influence = minor_influence_gain
		}
		hidden_effect = {
			if = {
				limit = {
					has_game_rule = bannermen_mode_enabled
					exists = scope:recipient.liege
					scope:attacker = {
						interactive_bannermen_is_rebel_trigger = {
							LIEGE = scope:defender
						}
					}
				}
				if = {
					limit = {
						is_attacker_in_war = scope:war
					}
					interactive_bannermen_civilwar_rebel_toast_effect = {
						LIEGE = scope:recipient.liege
						VASSAL = scope:recipient
						WAR = scope:war
					}
				}
				else_if = {
					limit = {
						is_defender_in_war = scope:war
					}
