advanced_torture.6101 = {
 type = character_event
 hidden = yes

 immediate = {
  if = {
   limit = {
    exists = scope:advanced_torture_prisoner

    scope:advanced_torture_prisoner = {
     is_alive = yes
     has_character_flag = advanced_torture_long_sleep_active
    }
   }

   if = {
    limit = {
     scope:advanced_torture_prisoner = {
      has_character_flag = advanced_torture_long_sleep_extend_3
     }
    }

    scope:advanced_torture_prisoner = {
     remove_character_flag = advanced_torture_long_sleep_extend_3
    }

    trigger_event = {
     id = advanced_torture.6101
     days = 3
    }
   }

   else = {
    advanced_torture_finish_long_sleep_deprivation_effect = yes

    if = {
     limit = {
      has_character_flag = advanced_torture_last_torture_killed_prisoner
     }

     advanced_torture_clear_torture_death_result_effect = yes

     trigger_event = {
      id = advanced_torture.1008
     }
    }

    else = {
     trigger_event = {
      id = advanced_torture.6201
     }
    }
   }
  }
 }
}
