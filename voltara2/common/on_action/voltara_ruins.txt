on_death = { on_actions = { voltara_ruins_patron_death } }
voltara_ruins_patron_death = { effect = {
 every_in_global_list = { variable = voltara_ruined_counties
  limit = { var:voltara_expedition_owner ?= root }
  remove_variable = voltara_expedition_owner
  if = { limit = { NOT = { holder = global_var:voltara_ruins_keeper } }
   voltara_set_abandoned_identity = yes
   title_province = { set_holding_type = voltara_ruin_holding }
   remove_county_modifier = voltara_settlement_county add_county_modifier = { modifier = voltara_abandoned_county }
   create_title_and_vassal_change = { type = granted save_scope_as = abandoned_change add_claim_on_loss = no }
   change_title_holder = { holder = global_var:voltara_ruins_keeper change = scope:abandoned_change take_baronies = yes }
   resolve_title_and_vassal_change = scope:abandoned_change
  }
 }
} }
