on_death = { on_actions = { voltara_dragon_death_cleanup } }
voltara_dragon_death_cleanup = { effect = {
 if = { limit = { has_trait = dragon }
 var:current_rider ?= { remove_variable = current_dragon remove_trait = dragonrider voltara_clear_deployment = yes }
 remove_variable = current_rider remove_list_global_variable = { name = voltara_living_dragons target = this }
 }
 else_if = { limit = { exists = var:current_dragon }
 var:current_dragon = { remove_variable = current_rider remove_character_flag = owned_dragon }
 remove_variable = current_dragon }
} }

on_army_monthly = { on_actions = { voltara_dragon_owner_month } }
voltara_dragon_owner_month = { trigger = { OR = { has_trait = dragonrider has_character_modifier = voltara_dragon_warfare } } effect = {
 if = { limit = { NOT = { voltara_has_living_dragon = yes } } remove_variable = current_dragon remove_trait = dragonrider voltara_clear_deployment = yes }
 else_if = { limit = { OR = { NOT = { is_current_dragonrider = yes } is_at_war = no NOT = { has_character_flag = voltara_dragon_deployed } } } voltara_clear_deployment = yes }
} }

on_game_start_after_lobby = { on_actions = { voltara_repair_existing_dragon_bonds } }
voltara_repair_existing_dragon_bonds = { effect = {
 every_in_global_list = { variable = voltara_living_dragons
 if = { limit = { exists = var:current_rider }
 save_scope_as = repair_dragon
 if = { limit = { var:current_rider = { is_alive = yes var:current_dragon ?= scope:repair_dragon } }
 var:current_rider = { add_trait = dragonrider set_relation_agot_dragon = { reason = voltara_dragon_bond_reason target = scope:repair_dragon } }
 }
 else = { var:current_rider = { remove_relation_agot_dragon = scope:repair_dragon } remove_variable = current_rider remove_character_flag = owned_dragon }
 }
 }
} }

 }
