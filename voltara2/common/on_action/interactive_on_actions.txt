				value = yes
			}
		}
		if = {
			limit = {
				always = no
			}
			set_global_variable = {
				name = AGOT_is_loaded
				value = yes
			}
			random_living_character = {
				set_variable = {
					name = current_rider
					value = yes
				}
				set_variable = {
					name = agot_pwl_direct
					value = yes
				}
				add_character_flag = owned_dragon
				add_to_variable_list = {
					name = kingsguard_members
					target = this
				}
				save_temporary_scope_as = miv_enemy
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_100
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 100
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_200
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 200
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_300
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 300
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_every_100
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 100
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_every_200
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 200
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_every_300
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 300
			}

		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_400
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 400
			}
		}
		if = {
			limit = {
				has_game_rule = scourge_of_the_gods_miv_enabled_every_400
			}
			set_global_variable = {
				name = miv_scourge_waiting
				value = yes
				years = 400
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = shattered_world_miv_enabled
					has_game_rule = diverse_world_miv_enabled
				}
			}
			every_ruler = {
				limit = {
					interactive_map_generation_placeholder_trigger = yes
				}
				add_to_list = miv_map_generation_placeholders
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = shattered_world_miv_enabled
					has_game_rule = divided_china_miv_enabled
				}
				exists = situation:dynastic_cycle
				exists = title:h_china
				exists = title:h_china.holder
			}
			situation:dynastic_cycle = {
				situation_top_sub_region = {
					change_phase = {
						phase = situation_dynastic_cycle_phase_chaos
					}
				}
			}
			if = {
				limit = {
					has_game_rule = divided_china_miv_enabled
				}
				interactive_divided_china_effect = yes
			}
		}

			limit = {
				OR = {
					has_game_rule = divided_japan_miv_enabled
					has_game_rule = shattered_world_miv_enabled
				}
				exists = title:e_japan
				exists = title:e_japan.holder
			}
			interactive_divided_japan_effect = yes
		}
		if = {
			limit = {
				has_game_rule = shattered_world_miv_enabled
			}
			every_ruler = {
				limit = {
					interactive_map_generation_ruler_trigger = yes
				}
				save_scope_as = ruler_to_update
				scope:ruler_to_update = {
					every_in_list = {
						list = miv_shatter_tributaries
						remove_from_list = miv_shatter_tributaries
					}
					every_tributary = {
						limit = {
							highest_held_title_tier >= tier_county
						}
						add_to_list = miv_shatter_tributaries
					}
					every_in_list = {
						list = miv_shatter_tributaries
						end_tributary = yes
					}
				}
				every_in_list = {
					list = miv_shatter_counties
					remove_from_list = miv_shatter_counties
				}
				every_held_title = {
					limit = {
						tier = tier_county
						NOT = {
							this = scope:ruler_to_update.capital_county
						}
					}
					add_to_list = miv_shatter_counties
				}
				every_in_list = {
					list = miv_shatter_counties
					save_scope_as = title_to_update
					scope:ruler_to_update = {
						create_character = {
							random_traits = yes
							age = { 20 30 }
							gender_female_chance = 5
							location = scope:ruler_to_update.capital_province
							culture = scope:title_to_update.culture
							faith = scope:title_to_update.faith
							save_scope_as = created_character
						}
						create_title_and_vassal_change = {
							type = created
							save_scope_as = change
							add_claim_on_loss = no
						}
						scope:title_to_update = {
							change_title_holder = {
								holder = scope:created_character
								change = scope:change
							}
						}
						resolve_title_and_vassal_change = scope:change
						remove_claim = scope:title_to_update
					}
				}
				every_in_list = {
					list = miv_shatter_titles
					remove_from_list = miv_shatter_titles
				}
				every_held_title = {
					limit = {
						tier >= tier_duchy
					}
					add_to_list = miv_shatter_titles
				}
				every_in_list = {
					list = miv_shatter_titles
					scope:ruler_to_update = {
						destroy_title = prev
					}
				}
				every_in_list = {
					list = miv_shatter_claims
					remove_from_list = miv_shatter_claims
				}
				every_claim = {
					add_to_list = miv_shatter_claims
				}
				every_in_list = {
					list = miv_shatter_claims
					scope:ruler_to_update = {
						remove_claim = prev
					}
				}
			}
		}
		if = {
			limit = {
				has_game_rule = diverse_world_miv_enabled
				has_game_rule = shattered_world_miv_enabled
			}
			interactive_random_ruler_placement_effect = yes
			every_ruler = {
				limit = {
					interactive_map_generation_ruler_trigger = yes
				}
				every_held_title = {
					limit = {
						tier = tier_county
					}
					set_county_culture = holder.culture
					set_county_faith = holder.faith
				}
			}
		}
		if = {
			limit = {
				has_game_rule = shattered_world_miv_enabled
			}
			every_religion_global = {
				every_faith = {
					limit = {
						exists = religious_head
					}
					remove_religious_head_title = yes
				}
			}
		}
	}
}

interactive_war_started = {
	effect = {
		if = {
			limit = {
				interactive_improved_civil_wars_trigger = yes
				NOT = {
					scope:war = {
						interactive_agot_megawar_trigger = yes
					}
				}
				scope:attacker = {
					interactive_bannermen_is_rebel_trigger = {
						LIEGE = scope:defender
					}
					is_landed = yes
				}
				OR = {
					has_game_rule = bannermen_mode_enabled
					debug_only = yes
					any_player = {
						OR = {
							in_diplomatic_range = scope:defender
							in_diplomatic_range = scope:attacker
						}
					}
				}
				NOT = {
					interactive_start_date = current_date
				}
				trigger_if = {
					limit = {
						has_game_rule = miv_performance_mode_full
						has_game_rule = bannermen_mode_enabled
					}
					any_player = {
						OR = {
							in_diplomatic_range = scope:defender
							in_diplomatic_range = scope:attacker
						}
					}
				}
				scope:war = {
					OR = {
						AND = {
							NOT = { using_cb = depose_war }
							NOT = { using_cb = refused_liege_demand_war }
						}
						any_player ={
							OR = {
								is_vassal_or_below_of = scope:defender
								this = scope:defender
							}
						}
					}
				}
			}
			scope:defender = {
				every_vassal = {
					limit = {
						interactive_bannermen_civilwar_eligible_trigger = {
							LIEGE = scope:defender
							LEADER = scope:attacker
						}
					}
					if = {
						limit = {
							has_game_rule = bannermen_mode_enabled
						}
						trigger_event = {
							id = interactive.0007
							days = { 10 30 }
						}
					}
					else = {
						trigger_event = {
							id = interactive.0007
							days = { 2 90 }
						}
					}
				}
				interactive_bannermen_civilwar_far_effect = {
					LIEGE = scope:defender
					LEADER = scope:attacker
					WAR = scope:war
				}
			}
			if = {
				limit = {
					has_game_rule = bannermen_mode_enabled
				}
				scope:defender = {
					every_vassal = {
						limit = {
							interactive_bannermen_loyalist_faction_trigger = yes
							interactive_bannermen_civilwar_eligible_trigger = {
								LIEGE = scope:defender
								LEADER = scope:attacker
							}
							interactive_bannermen_fit_trigger = yes
							NOR = {
								is_attacker_in_war = scope:war
								is_defender_in_war = scope:war
							}
						}
						add_to_list = miv_cw_loyalists
					}
				}
				every_in_list = {
					list = miv_cw_loyalists
					scope:war = {
						add_defender = prev
					}
				}
				scope:war = {
					every_war_defender = {
						limit = {
							interactive_bannermen_is_rebel_trigger = {
								LIEGE = scope:defender
							}
						}
						save_temporary_scope_as = miv_loyal_vassal
						interactive_bannermen_batch_toast_effect = {
							KEY = loyal
							EVENT = 0017
							HOLDER = scope:defender
							WAR = scope:war
							SIDE_OF = scope:defender
							VASSAL = scope:miv_loyal_vassal
						}
					}
				}
			}
			if = {
				limit = {
					OR = {
						has_game_rule = civil_war_indirect_vassals_leader
						has_game_rule = civil_war_indirect_vassals_enabled
					}
					has_game_rule = bannermen_mode_enabled
				}
				scope:war = {
					save_scope_as = miv_cw_war
				}
				scope:war = {
					every_war_attacker = {
						limit = {
							interactive_bannermen_is_rebel_trigger = {
								LIEGE = scope:defender
							}
							trigger_if = {
								limit = {
									has_game_rule = civil_war_indirect_vassals_leader
								}
								this = scope:war.primary_attacker
							}
						}
						interactive_bannermen_civilwar_core_cascade_effect = yes
					}
					if = {
						limit = {
							has_game_rule = civil_war_indirect_vassals_enabled
						}
						every_war_defender = {
							limit = {
								interactive_bannermen_is_rebel_trigger = {
									LIEGE = scope:defender
								}
							}
							interactive_bannermen_civilwar_core_cascade_effect = yes
						}
					}
				}
			}
			if = {
				limit = {
					has_game_rule = bannermen_mode_enabled
				}
				scope:war = {
					set_variable = miv_civil_war
				}
				scope:attacker = {
					save_scope_as = VASSAL
					set_variable = {
						name = miv_fellow_rebels
						value = 0
						days = 30
					}
				}
				scope:war = {
					every_war_attacker = {
						limit = {
							interactive_bannermen_is_rebel_trigger = {
								LIEGE = scope:defender
							}
						}
						save_temporary_scope_as = miv_rebel
						if = {
							limit = {
								NOT = {
									liege = {
										is_attacker_in_war = scope:war
									}
								}
							}
							interactive_bannermen_brand_oathbreaker_effect = {
								LIEGE = scope:miv_rebel.liege
								VASSAL = scope:miv_rebel
							}
						}
						if = {
							limit = {
								NOT = {
									this = scope:VASSAL
								}
							}
							scope:VASSAL = {
								change_variable = {
									name = miv_fellow_rebels
									add = 1
								}
							}
						}
						if = {
							limit = {
								exists = scope:war.primary_attacker.joined_faction
								interactive_bannermen_is_rebel_trigger = {
									LIEGE = scope:war.primary_defender
								}
							}
							if = {
								limit = {
									exists = joined_faction
									NOT = {
										joined_faction = scope:war.primary_attacker.joined_faction
									}
								}
								leave_faction = joined_faction
							}
							join_faction_skip_check = scope:war.primary_attacker.joined_faction
						}
					}
				}
				if = {
					limit = {
						scope:VASSAL = {
							exists = var:miv_fellow_rebels
							var:miv_fellow_rebels >= 1
						}
					}
					save_temporary_scope_value_as = {
						name = miv_toast_grouped
						value = yes
					}
					interactive_bannermen_toast_war_effect = {
						WAR = scope:war
						SIDE_OF = scope:defender
						VASSAL = scope:VASSAL
						TYPE = msg_war_oathbreaker
						TITLE = miv_toast_rebel_many
						TOOLTIP = miv_toast_rebel_many_desc
					}
				}
				else = {
					interactive_bannermen_toast_war_effect = {
						WAR = scope:war
						SIDE_OF = scope:defender
						VASSAL = scope:VASSAL
						TYPE = msg_war_oathbreaker

interactive_bannermen = {
	effect = {
		if = {
			limit = {
				always = no
				NOT = {
					has_game_rule = bannermen_mode_enabled
				}
				OR = {
					debug_only = yes
					any_player = {
						OR = {
							in_diplomatic_range = scope:defender
							in_diplomatic_range = scope:attacker
						}
					}
				}
				NOT = {
					interactive_start_date = current_date
				}
				war = {
					is_civil_war = no
				}
				NOT = {
					scope:attacker = {
						is_vassal_or_below_of = scope:defender
					}
				}
			}
			scope:defender = {
				every_vassal = {
					limit = {
						NOT = {
							scope:defender = {
								has_character_flag = dont_call_vassals
							}
						}
						OR = {
							AND = {
								scope:defender = {
									highest_held_title_tier < tier_empire
								}
								NOT = {
									scope:defender = {
										any_held_title = {
											count >= 2
											tier = tier_kingdom
										}
									}
								}
								highest_held_title_tier > tier_barony
							}
							highest_held_title_tier > tier_county
						}
						max_military_strength > interactive_minimum_vassal_strength
						is_landed = yes
					}
					if = {
						limit = {
							opinion = {
								target = scope:defender
								value > 15
							}
							NOT = {
								is_at_war_with = scope:attacker
							}
							NOT = {
								is_at_war_with = scope:defender
							}
						}
						scope:war = {
							add_defender = prev
						}
						if = {
							limit = {
								NOT = {
									has_game_rule = war_exhaustion_miv_disabled
								}
							}
							add_opinion = {
								target = scope:defender
								modifier = war_exhaustion_vassal_opinion
								opinion = -10
							}
						}
					}
				}
			}
			scope:attacker = {
				every_vassal = {
					limit = {
						NOT = {
							scope:attacker = {
								has_character_flag = dont_call_vassals
							}
						}
						OR = {
							AND = {
								scope:attacker = {
									highest_held_title_tier < tier_empire
								}
								NOT = {
									scope:attacker = {
										any_held_title = {
											count >= 2
											tier = tier_kingdom
										}
									}
								}
								highest_held_title_tier > tier_barony
							}
							highest_held_title_tier > tier_county
						}
						max_military_strength > interactive_minimum_vassal_strength
						is_landed = yes
					}
					if = {
						limit = {
							opinion = {
								target = scope:attacker
								value > 20
							}
							NOT = {
								is_at_war_with = scope:defender
							}
							NOT = {
								is_at_war_with = scope:attacker
							}
						}
						scope:war = {
							add_attacker = prev
						}
						if = {
							limit = {
								NOT = {
									has_game_rule = war_exhaustion_miv_disabled
								}
							}
							add_opinion = {
								target = scope:attacker
								modifier = war_exhaustion_vassal_opinion
								opinion = -10
							}
						}
					}
				}
			}
		}
	}
}

interactive_war_participants = {
	effect = {
		if = {
			limit = {
				always = no
				OR = {
					has_game_rule = bannermen_mode_enabled
					debug_only = yes
					any_player = {
						OR = {
							in_diplomatic_range = scope:war.primary_defender
							in_diplomatic_range = scope:war.primary_attacker
						}
					}
				}
				scope:war = {
					is_civil_war = no
				}
				NOT = {
					scope:war.primary_attacker = {
						is_vassal_or_below_of = scope:war.primary_defender
					}
				}
				NOT = {
					interactive_start_date = current_date
				}
			}
			scope:war = {
				every_war_participant = {
					limit = {
						OR = {
							is_vassal_of = scope:war.primary_attacker
							interactive_bannermen_is_rebel_trigger = {
								LIEGE = scope:war.primary_defender
							}
						}
					}
					save_scope_as = participant
					scope:participant = {
						every_ally = {
							limit = {
								is_vassal_of = scope:participant
								NOT = {
									scope:participant = {
										has_character_flag = dont_call_vassals
									}
								}
								highest_held_title_tier > tier_barony
								max_military_strength > interactive_minimum_vassal_strength
								is_landed = yes
								opinion = {
									target = scope:participant
									value >= 80
								}
							}
							if = {
								limit = {
									NOT = {
										is_at_war_with = scope:war.primary_attacker
									}
									NOT = {
										is_at_war_with = scope:war.primary_defender
									}
								}
								if = {
									limit = {
										scope:war = {
											any_war_defender = {
												this = scope:participant
											}
										}
									}
									add_to_list = miv_join_defender
									if = {
										limit = {
											NOT = {
												has_game_rule = war_exhaustion_miv_disabled
											}
										}
										add_opinion = {
											target = scope:participant
											modifier = war_exhaustion_vassal_opinion
											opinion = -5
										}
									}
								}
								else_if = {
									limit = {
										scope:war = {
											any_war_attacker = {
												this = scope:participant
											}
										}
									}
									add_to_list = miv_join_attacker
									if = {
										limit = {
											NOT = {
												has_game_rule = war_exhaustion_miv_disabled
											}
										}
										add_opinion = {
											target = scope:participant
											modifier = war_exhaustion_vassal_opinion
											opinion = -5
										}
									}
								}
							}
						}
					}
				}
			}
			every_in_list = {
				list = miv_join_defender
				scope:war = {
					add_defender = prev
				}
			}
			every_in_list = {
				list = miv_join_attacker
				scope:war = {
					add_attacker = prev
				}
			}
		}
	}
}

miv_accolade_artifact_holder_lost = {
	trigger = {
		exists = this
		OR = {
			has_variable = miv_accolade_artifact_weapon_artifact
			has_variable = miv_accolade_artifact_armor_artifact
		}
	}
	effect = {
		var:miv_accolade_artifact_weapon_artifact ?= {
			if = {
				limit = {
					miv_accolade_artifact_slain_capture_trigger = yes
				}
				miv_accolade_artifact_steal_item_effect = yes
			}
		}
		var:miv_accolade_artifact_armor_artifact ?= {
			if = {
				limit = {
					miv_accolade_artifact_slain_capture_trigger = yes
				}
				miv_accolade_artifact_steal_item_effect = yes
			}
		}
		if = {
			limit = {
				miv_accolade_artifact_enabled_trigger = yes
			}
			miv_accolade_artifact_safekeep_accolade_effect = yes
		}
		else = {
			miv_accolade_artifact_recall_weapon_effect = yes
			miv_accolade_artifact_recall_armor_effect = yes
		}
		var:miv_accolade_artifact_weapon_artifact ?= {
			miv_accolade_artifact_queue_repair_effect = yes
