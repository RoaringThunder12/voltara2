# SECRET WARRIOR TRAINING
secret_warrior_training_is_valid_trigger = {
	$OWNER$ = {
		NOT = {
			has_trait = valor_child_prowess
			has_trait = education_martial_prowess
		}
	}
}

secret_warrior_training_is_shunned_trigger = {
	$OWNER$ = {
		# Male Dominated
		AND = {
			is_female = yes
			OR = {
				has_realm_law = male_only_law
				has_realm_law = male_preference_law
				liege_or_court_owner = { # any_liege_or_above = {
					has_realm_law = male_only_law
					has_realm_law = male_preference_law
				}
			}
		}
		# Female Dominated
		AND = {
			is_female = no
			OR = {
				has_realm_law = female_only_law
				has_realm_law = female_preference_law
				liege_or_court_owner = { # any_liege_or_above = {
					has_realm_law = female_only_law
					has_realm_law = female_preference_law
				}
			}
		}
	} # Owner Scope
}
