							target = $VASSAL$.capital_province

						}
					}

						target = $VASSAL$.capital_province
						value <= miv_vassal_distance_value
					}
				}
			}
		}

		limit = {
			has_game_rule = vassal_distance_land_or_sea
		}
		any_sub_realm_county = {
			any_title_to_title_neighboring_and_across_water_county = {

					}
				}
			}
		}
