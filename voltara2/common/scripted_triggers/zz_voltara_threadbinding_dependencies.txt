can_increase_lifespan_trigger = {
	exists = var:mastered_schools
	OR = {
		AND = {
			var:mastered_schools >= 1
			NOR = {
				has_trait = lifespan
				has_trait = lifespan_magic
			}
		}
		AND = {
			var:mastered_schools >= 1
			has_trait = lifespan
		}
		AND = {
			var:mastered_schools >= 1
			has_trait = lifespan_magic_1
		}
		AND = {
			var:mastered_schools >= 2
			has_trait = lifespan_magic_2
		}
		AND = {
			var:mastered_schools >= 3
			has_trait = lifespan_magic_3
		}
		AND = {
			var:mastered_schools >= 4
			has_trait = lifespan_magic_4
		}
		AND = {
			var:mastered_schools >= 5
			has_trait = lifespan_magic_5
		}
	}
}

trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = {
	OR = {
		AND = {
			faith = { #My faith doesn't approve
				save_temporary_scope_as = check_faith
			}
			trait_is_shunned_or_criminal_in_faith_trigger = { TRAIT = $TRAIT$ FAITH = scope:check_faith GENDER_CHARACTER = $GENDER_CHARACTER$ }
		}
		AND = { #Liege's faith doesn't approve
			liege ?= {
				faith = {
					save_temporary_scope_as = check_faith
				}
			}
			trait_is_shunned_or_criminal_in_faith_trigger = { TRAIT = $TRAIT$ FAITH = scope:check_faith GENDER_CHARACTER = $GENDER_CHARACTER$ }
		}
	}
}
