advanced_torture_can_torture_prisoner_trigger = {
 exists = scope:advanced_torture_prisoner

 scope:advanced_torture_prisoner = {
  is_alive = yes
  is_imprisoned = yes
  NOT = { has_character_modifier = advanced_torture_cooldown }
 }
}

advanced_torture_can_pay_light_torture_trigger = {
 OR = {
  prestige >= 150

  AND = {
   has_trait = sadistic
   prestige >= 75
  }
 }
}

   prestige >= 140
  }
