mongol_demand_subjugation_interaction = {
is_valid = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
}

	category = interaction_category_hostile
	icon = icon_dismiss_temporary_regency

	auto_accept = yes
	use_diplomatic_range = yes

	desc = mongol_demand_subjugation_interaction_desc
	
	is_available = {
		highest_held_title_tier >= tier_empire
		any_owned_story = {
			type = story_mongol_invasion
		}
		top_liege ?= this
	}

	is_shown = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }

		scope:recipient = {
			this != scope:actor
			is_landless_adventurer = no
			top_liege ?= this
			any_neighboring_top_liege_realm_owner = {
				this = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		OR = {
			scope:recipient = {
				highest_held_title_tier <= tier_kingdom
			}
			scope:actor = {
				is_target_in_variable_list = {
					name = subjugation_offer_under_consideration
					target = scope:recipient
				}
			}
		}
	}

	on_accept = {
		custom_tooltip = mongol_demand_subjugation_interaction_tooltip
		hidden_effect = {
			scope:actor = {
				save_scope_as = mongol_emperor
			}
			scope:recipient = {
				send_mongol_subjugation_demand_effect = yes
			}
		}
	}
}
