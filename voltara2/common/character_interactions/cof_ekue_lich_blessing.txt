cof_ekue_ascend_lich_interaction = {
is_valid = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
}

	common_interaction = yes
	category = interaction_category_magic
	desc = cof_ekue_ascend_lich_interaction_desc
	icon = icon_thralldom

	is_shown = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }

		NOT = { scope:actor = scope:recipient }
		scope:actor = {
				has_perk = necromancy_master_perk
				magicka_max > 60
			}
		scope:recipient = {
					has_trait = vt_ek_undead_character
				}
	}

	is_valid_showing_failures_only = {
		custom_tooltip = {
			text = cof_ekue_magicka_60
			scope:actor = {
				var:magicka > 60
			}
		}
	}

	ai_potential = {
		always = no
	}
	
	ai_will_do = {
		base = 0
	}

	on_accept = {
		scope:actor = {
			add_hook = {
				type = ascended_undead_hook
				target = scope:recipient
			}

			add_magicka = { VALUE = 60.negate }
			change_variable = { name = magicka add = -60 }
		}

		scope:recipient = {
			remove_trait = vt_ek_undead_character
			add_trait = vt_ek_lich_character

			add_opinion = {
				modifier = cof_ekue_ascend_lich_opinion
				target = scope:actor
				years = 100
			}
		}

		custom_tooltip = {
			text = cof_ekue_ascend_lich_interaction_tp
		}
	}
	
	auto_accept = yes
}
