voltara_sell_slave_interaction = {
 category = interaction_category_friendly icon = icon_gold common_interaction = yes
 desc = voltara_sell_slave_interaction_desc
 populate_actor_list = { scope:actor = { every_courtier = { limit = { voltara_slave_can_be_sold = yes } add_to_list = characters } } }
 is_shown = { scope:actor = { NOT = { voltara_is_ruins_placeholder = yes } } scope:recipient = { NOT = { voltara_is_ruins_placeholder = yes } }
  scope:actor != scope:recipient
  scope:actor = { voltara_slavery_is_legal = yes any_courtier = { voltara_slave_can_be_sold = yes } }
  scope:recipient = { voltara_slavery_is_legal = yes }
 }
 is_valid_showing_failures_only = {
  scope:actor = { in_diplomatic_range = scope:recipient }
  scope:recipient = { gold >= 30 is_alive = yes NOT = { is_at_war_with = scope:actor } }
 }
 can_send = { exists = scope:secondary_actor scope:secondary_actor = { voltara_slave_can_be_sold = yes } }
 on_send = { scope:secondary_actor = { add_character_flag = { flag = voltara_slave_sale_pending days = 90 } } }
 on_decline = { scope:secondary_actor = { remove_character_flag = voltara_slave_sale_pending } }
 on_accept = {
  scope:secondary_actor = { remove_character_flag = voltara_slave_sale_pending }
  if = { limit = {
   scope:secondary_actor = { voltara_slave_can_be_sold = yes }
   scope:actor = { voltara_slavery_is_legal = yes is_alive = yes }
   scope:recipient = { voltara_slavery_is_legal = yes is_alive = yes gold >= 30 }
  }
   scope:recipient = { remove_short_term_gold = 30 add_trait = voltara_slaver }
   scope:actor = { add_gold = 30 add_trait = voltara_slaver }
   scope:secondary_actor = { set_variable = { name = voltara_slave_owner value = scope:recipient } }
   scope:recipient = { add_courtier = scope:secondary_actor }
  }
 }
 ai_accept = { base = 10 modifier = { add = -100 scope:recipient = { has_trait = compassionate } } modifier = { add = 20 scope:recipient = { has_trait = voltara_slaver } } }
 ai_targets = { ai_recipients = neighboring_rulers }
 ai_frequency = 24
 ai_will_do = { base = 10 }
}
