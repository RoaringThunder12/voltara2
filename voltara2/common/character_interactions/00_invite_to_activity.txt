invite_to_activity_interaction = {
is_valid = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
}

	category = interaction_category_diplomacy
	greeting = positive
	notification_text = invite_to_activity_flavor
	reply_item_key = invite_to_activity_reply_item_key
	icon = activity_feast
	common_interaction = yes
	desc = invite_to_activity_interaction_desc

	ai_maybe = yes
	can_send_despite_rejection = yes
	
	is_highlighted = {
		always = yes
	}

	is_shown = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }

		scope:puppet_or_actor.involved_activity ?= {
			OR = {
				has_activity_type = activity_tournament
				has_activity_type = activity_wedding
				has_activity_type = activity_feast
				has_activity_type = activity_hunt
				has_activity_type = activity_chariot_race
			}
			is_current_phase_active = no
			activity_host ?= scope:puppet_or_actor
		}
		scope:recipient = { can_join_activity = scope:puppet_or_actor.involved_activity }
	}

	on_send = {
		scope:recipient = {
			invite_to_activity = scope:puppet_or_actor.involved_activity
		}
	}
	
	on_accept = {
		# Acceptance handled by the activity system itself
		scope:puppet_or_actor = {
			send_interface_message = {
				type = event_activity_good_with_text
				title = invite_to_activity_accepted
				desc = invite_to_activity_accepted_desc
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
				scope:recipient = {
					accept_activity_invite = scope:puppet_or_actor.involved_activity
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_message = {
					type = event_activity_good_with_text
					title = invite_to_activity_accepted_puppeteer
					desc = invite_to_activity_accepted_desc_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
	}

	on_decline = {
		# Acceptance handled by the activity system itself
		scope:puppet_or_actor = {
			send_interface_message = {
				type = event_activity_bad_with_text
				title = invite_to_activity_refused
				desc = invite_to_activity_refused_desc
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
				scope:recipient = {
					decline_activity_invite = scope:puppet_or_actor.involved_activity
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_message = {
					type = event_activity_bad_with_text
					title = invite_to_activity_refused_puppeteer
					desc = invite_to_activity_refused_desc_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
	}

	ai_accept = {
		base = 0

		## You should not add modifiers here! They won't show up in the UI
		# If you want to add modifiers, they should be added to base_activity_modifier in 00_activity_scripted_modifiers.txt
		# Any new activity should use base_activity_modifier = yes in their activity to get these default modifiers

		activity_modifier = {
			object = scope:puppet_or_actor.involved_activity
			target = scope:recipient			
		}
	}
}
