interactive_ask_ally_join_faction = {
is_valid = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
}

	common_interaction = yes
	interface_priority = 68
	category = interaction_category_diplomacy
	desc = interactive_ask_ally_join_faction_desc
	icon = alliance
	force_notification = yes
	greeting = positive
	notification_text = FORCE_JOIN_FACTION_NOTIFICATION
	ai_min_reply_days = 0
	ai_max_reply_days = 0
	is_shown = {
 # Voltara: ruins are colonized through the county panel, not character diplomacy.
 scope:actor = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:recipient = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }

		NOT = {
			has_game_rule = miv_invite_faction_disabled
		}
		scope:actor = {
			exists = liege
			exists = joined_faction
			NOT = {
				is_at_war_with = liege
			} 
		}
		scope:recipient = {
			is_allied_to = scope:actor
			exists = liege
			is_vassal_of = scope:actor.liege
			NOT = {
				exists = joined_faction
			}
			NOT = {
				is_at_war_with = scope:actor
			}
			NOT = {
				is_at_war_with = scope:actor.liege
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			can_join_faction = scope:actor.joined_faction
		}
	}
	on_accept = {
		scope:actor = {
			if = {
				limit = {
					exists = joined_faction
				}
				show_as_tooltip = {
					scope:recipient = {
						join_faction = scope:actor.joined_faction
					}
				}
				hidden_effect = {
					send_interface_message = {
						type = event_faction_neutral
						title = force_join_faction_interaction_notification
						right_icon = scope:recipient
						scope:recipient = {
							join_faction_forced = {
								faction = scope:actor.joined_faction
								forced_by = scope:actor
								years = 5
							}
						}
					}
				}
			}
		}
	}
	on_decline = {
	}
	ai_accept = {
		base = -20
		opinion_modifier = {
			opinion_target = scope:actor
		}
		opinion_modifier = {
			opinion_target = scope:recipient.liege
			multiplier = -1
		}
	}
	ai_will_do = {
		base = 0
	}
}
