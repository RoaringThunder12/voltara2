advanced_torture_initialize_sadism_threshold_effect = {
 if = {
  limit = {
   NOT = { has_character_flag = advanced_torture_sadism_threshold_set }
  }

  add_character_flag = advanced_torture_sadism_threshold_set

  random_list = {
   20 = {
    add_character_flag = advanced_torture_sadism_threshold_3
   }

   20 = {
    add_character_flag = advanced_torture_sadism_threshold_4
   }

   20 = {
    add_character_flag = advanced_torture_sadism_threshold_5
   }

   20 = {
    add_character_flag = advanced_torture_sadism_threshold_6
   }

   20 = {
    add_character_flag = advanced_torture_sadism_threshold_7
   }
  }
 }
}

advanced_torture_check_sadism_progression_effect = {
 if = {
  limit = {
   NOT = { has_trait = sadistic }
   NOT = { has_character_flag = advanced_torture_sadism_trait_event_fired }

   OR = {
    AND = {
     has_character_flag = advanced_torture_sadism_threshold_3
     var:advanced_torture_strong_tortures >= 3
    }

    AND = {
     has_character_flag = advanced_torture_sadism_threshold_4
     var:advanced_torture_strong_tortures >= 4
    }

    AND = {
     has_character_flag = advanced_torture_sadism_threshold_5
     var:advanced_torture_strong_tortures >= 5
    }

    AND = {
     has_character_flag = advanced_torture_sadism_threshold_6
     var:advanced_torture_strong_tortures >= 6
    }

    AND = {
     has_character_flag = advanced_torture_sadism_threshold_7
     var:advanced_torture_strong_tortures >= 7
    }
   }
  }

  add_character_flag = advanced_torture_sadism_trait_event_fired

  trigger_event = {
   id = advanced_torture.5500
  }
 }
}



advanced_torture_initialize_torturer_threshold_effect = {
 if = {
  limit = {
   NOT = { has_trait = torturer }
   NOT = { has_character_flag = advanced_torture_torturer_threshold_set }
  }

  add_character_flag = advanced_torture_torturer_threshold_set

  random_list = {
   10 = { add_character_flag = advanced_torture_torturer_threshold_4 }
   15 = { add_character_flag = advanced_torture_torturer_threshold_5 }
   20 = { add_character_flag = advanced_torture_torturer_threshold_6 }
   20 = { add_character_flag = advanced_torture_torturer_threshold_7 }
   15 = { add_character_flag = advanced_torture_torturer_threshold_8 }
   10 = { add_character_flag = advanced_torture_torturer_threshold_9 }
   10 = { add_character_flag = advanced_torture_torturer_threshold_10 }
  }
 }
}

advanced_torture_check_torturer_progression_effect = {
 if = {
  limit = {
   NOT = { has_trait = torturer }
   has_variable = advanced_torture_tortures_total
  }

  advanced_torture_initialize_torturer_threshold_effect = yes

  if = {
   limit = {
    OR = {
     AND = { has_character_flag = advanced_torture_torturer_threshold_4 var:advanced_torture_tortures_total >= 4 }
     AND = { has_character_flag = advanced_torture_torturer_threshold_5 var:advanced_torture_tortures_total >= 5 }
     AND = { has_character_flag = advanced_torture_torturer_threshold_6 var:advanced_torture_tortures_total >= 6 }
     AND = { has_character_flag = advanced_torture_torturer_threshold_7 var:advanced_torture_tortures_total >= 7 }
     AND = { has_character_flag = advanced_torture_torturer_threshold_8 var:advanced_torture_tortures_total >= 8 }
     AND = { has_character_flag = advanced_torture_torturer_threshold_9 var:advanced_torture_tortures_total >= 9 }
     AND = { has_character_flag = advanced_torture_torturer_threshold_10 var:advanced_torture_tortures_total >= 10 }
    }
   }

   add_trait = torturer
  }
 }
}
