advanced_torture_apply_torture_opinion_light_effect = {
 if = {
  limit = { exists = scope:advanced_torture_prisoner }
  scope:advanced_torture_prisoner = {
   if = {
    limit = { NOT = { this = root } }
    add_opinion = {
     target = root
     modifier = advanced_torture_tortured_me_light_opinion
     years = 10
    }
   }
   every_close_family_member = {
    limit = { is_alive = yes NOT = { this = root } }
    add_opinion = {
     target = root
     modifier = advanced_torture_tortured_family_member_light_opinion
     years = 10
    }
   }
  }
 }
}

advanced_torture_apply_house_relation_damage_minor_effect = {
 if = {
  limit = {
   exists = house
   exists = scope:advanced_torture_prisoner
   scope:advanced_torture_prisoner = {
    exists = house
   }
  }

  house = {
   save_scope_as = advanced_torture_torturer_house
  }

  scope:advanced_torture_prisoner = {
   house = {
    save_scope_as = advanced_torture_prisoner_house
   }
  }

  scope:advanced_torture_torturer_house = {
   if = {
    limit = {
     NOT = { this = scope:advanced_torture_prisoner_house }
    }

    impact_house_relation = {
     target = scope:advanced_torture_prisoner_house
     steps = -0.10
     description = advanced_torture_house_relation_torture_minor_desc
     notification = yes
    }
   }
  }
 }
}
