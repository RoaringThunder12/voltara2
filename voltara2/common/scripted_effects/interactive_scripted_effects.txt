interactive_bannermen_summon_effect = {
	save_temporary_scope_value_as = {
		name = miv_score
		value = interactive_bannermen_muster_score
	}
	if = {
		limit = {
			NOT = {
				has_character_flag = miv_bannermen_summoned
			}
		}
		add_character_flag = {
			flag = miv_bannermen_summoned
			days = 1
		}
		if = {
			limit = {
				scope:miv_score >= interactive_bannermen_prompt_threshold
			}
			trigger_event = {
				id = interactive.0004
				days = { 10 30 }
			}
		}
		else_if = {
			limit = {
				scope:miv_score >= interactive_bannermen_reluctant_threshold
			}
			trigger_event = {
				id = interactive.0004
				days = { 60 90 }
			}
		}
		else = {
			trigger_event = {
				id = interactive.0004
				days = { 150 180 }
			}
		}
	}
	if = {
		limit = {
			NOT = {
				has_game_rule = war_exhaustion_miv_disabled
			}
			NOT = {
				scope:miv_score >= interactive_bannermen_prompt_threshold
			}
		}
		if = {
			limit = {
				scope:miv_score >= interactive_bannermen_reluctant_threshold
			}
			interactive_bannermen_toll_commit_effect = {
				TOLL = miv_toll_r1
				WAR = scope:war
			}
		}
		else = {
			interactive_bannermen_toll_commit_effect = {
				TOLL = miv_toll_r3
				WAR = scope:war
			}
		}
	}
}

interactive_bannermen_automatic_betrayal_effect = {
	save_temporary_scope_as = liege
	if = {
		limit = {
			highest_held_title_tier >= tier_duchy
			is_landed = yes
			is_ruler = yes
			scope:war = {
				interactive_bannermen_callable_war_trigger = {
					LIEGE = scope:liege
				}
			}
			OR = {
				is_ai = no
				scope:war = {
					interactive_bannermen_worth_calling_trigger = {
						LIEGE = scope:liege
					}
				}
			}
			trigger_if = {
				limit = {
					is_ai = yes
					has_game_rule = miv_performance_mode_full
				}
				any_player = {
					in_diplomatic_range = scope:liege
				}
			}
		}
		save_scope_as = miv_bannermen_liege
		every_vassal = {
			limit = {
				interactive_bannermen_scope_trigger = {
					LIEGE = scope:liege
				}
				interactive_bannermen_available_trigger = {
					LIEGE = scope:liege
				}
			}
			save_temporary_scope_as = vassal
			if = {
				limit = {
					miv_bannermen_distance_trigger = {
						LIEGE = scope:liege
						VASSAL = scope:vassal
					}
				}
				interactive_bannermen_refuse_effect = yes
			}
		}
	}
}

interactive_bannermen_refuse_effect = {
	save_temporary_scope_value_as = {
		name = miv_score
		value = interactive_bannermen_muster_score
	}
	if = {
		limit = {
			scope:miv_score < interactive_bannermen_betrayal_threshold
			exists = scope:miv_bannermen_liege
			NOT = {
				is_target_in_variable_list = {
					name = miv_betrayal_pending_lieges
					target = scope:miv_bannermen_liege
				}
			}
		}
		add_to_variable_list = {
			name = miv_betrayal_pending_lieges
			target = scope:miv_bannermen_liege
			days = 91
		}
		trigger_event = {
			id = interactive.0008
			days = { 60 90 }
		}
	}
}

interactive_bannermen_toast_side_effect = {
	$WAR$ = {
		save_temporary_scope_as = miv_toast_war
	}
	$SIDE_OF$ = {
		save_temporary_scope_as = miv_toast_anchor
	}
	$VASSAL$ = {
		save_scope_as = VASSAL
	}
	every_player = {
		limit = {
			OR = {
				NOT = {
					this = scope:VASSAL
				}
				exists = scope:miv_toast_grouped
			}
			is_participant_in_war = scope:miv_toast_war
			OR = {
				AND = {
					is_attacker_in_war = scope:miv_toast_war
					scope:miv_toast_anchor = {
						is_attacker_in_war = scope:miv_toast_war
					}
				}
				AND = {
					is_defender_in_war = scope:miv_toast_war
					scope:miv_toast_anchor = {
						is_defender_in_war = scope:miv_toast_war
					}
				}
			}
		}
		send_interface_toast = {
			type = $TYPE$
			title = $TITLE$
			left_icon = scope:VASSAL
			custom_tooltip = $TOOLTIP$
		}
	}
}

interactive_bannermen_preview_effect = {
	every_vassal = {
		limit = {
			interactive_bannermen_scope_trigger = {
				LIEGE = scope:liege
			}
		}
		save_temporary_scope_as = vassal
		save_temporary_scope_value_as = {
			name = miv_score
			value = interactive_bannermen_muster_score
		}
		if = {
			limit = {
				scope:liege = {
					is_ai = no
				}
			}
			set_variable = {
				name = miv_widget_sort
				value = scope:miv_score
				days = 1
			}
		}
		if = {
			limit = {
				interactive_bannermen_serving_trigger = {
					LIEGE = scope:liege
				}
			}
			add_to_list = interactive_bannermen_serving
		}
		if = {
			limit = {
				interactive_bannermen_rebellion_trigger = {
					LIEGE = scope:liege
				}
			}
			add_to_list = interactive_bannermen_rebellion
		}
		else_if = {
			limit = {
				NOT = {
					interactive_bannermen_available_trigger = {
						LIEGE = scope:liege
					}
				}
			}
			add_to_list = interactive_bannermen_unavailable
		}
		else = {
			if = {
				limit = {
					scope:miv_score >= interactive_bannermen_prompt_threshold
				}
				add_to_list = interactive_bannermen_prompt
			}
			else_if = {
				limit = {
					scope:miv_score >= interactive_bannermen_reluctant_threshold
				}
				add_to_list = interactive_bannermen_reluctant
			}
			else_if = {
				limit = {
					scope:miv_score >= interactive_bannermen_answer_threshold
				}
				add_to_list = interactive_bannermen_grudging
			}
			else_if = {
				limit = {
					scope:miv_score < interactive_bannermen_betrayal_threshold
				}
				add_to_list = interactive_bannermen_betray
			}
			else = {
				add_to_list = interactive_bannermen_refuse
			}
		}
	}
}

miv_accolade_artifact_sync_item_effect = {
	if = {
		limit = {
			miv_accolade_artifact_enabled_trigger = yes
			exists = var:miv_accolade_artifact_accolade
		}
		miv_accolade_artifact_protect_item_effect = yes
		miv_accolade_artifact_find_patron_effect = yes
		clear_saved_scope = miv_accolade_artifact_current_knight
		save_scope_as = miv_accolade_artifact_sync_item
		var:miv_accolade_artifact_accolade = {
			acclaimed_knight ?= {
				miv_accolade_artifact_select_knight_effect = yes
			}
		}
		if = {
			limit = {
				exists = scope:miv_accolade_artifact_next_knight
			}
			clear_saved_scope = miv_accolade_artifact_current_knight
			scope:miv_accolade_artifact_next_knight = {
				miv_accolade_artifact_select_knight_effect = yes
			}
		}
		if = {
			limit = {
				exists = scope:miv_accolade_artifact_current_patron
			}
			set_variable = {
				name = miv_accolade_artifact_patron
				value = scope:miv_accolade_artifact_current_patron
			}
			if = {
				limit = {
					exists = scope:miv_accolade_artifact_current_knight
				}
				if = {
					limit = {
						NOT = {
							artifact_owner = scope:miv_accolade_artifact_current_knight
						}
					}
					unequip_artifact_from_owner = yes
					set_owner = {
						target = scope:miv_accolade_artifact_current_knight
					}
				}
				if = {
					limit = {
						is_equipped = no
					}
					equip_artifact_to_owner_replace = yes
				}
			}
			else = {
				miv_accolade_artifact_safekeep_item_effect = yes
			}
		}
		else = {
			miv_accolade_artifact_release_item_effect = yes
		}
	}
	else = {
		miv_accolade_artifact_release_item_effect = yes
	}
}

miv_accolade_artifact_steal_item_effect = {
	save_scope_as = miv_accolade_artifact_stolen
	scope:old_acclaimed_knight = {
		save_scope_as = miv_accolade_artifact_slain_knight
	}
	artifact_owner = {
		save_scope_as = miv_accolade_artifact_slayer
		if = {
			limit = {
				exists = liege
			}
			liege = {
				save_scope_as = miv_accolade_artifact_taker
			}
		}
		else = {
			save_scope_as = miv_accolade_artifact_taker
		}
	}
	miv_accolade_artifact_find_patron_effect = yes
	if = {
		limit = {
			exists = scope:miv_accolade_artifact_current_patron
			scope:miv_accolade_artifact_taker = scope:miv_accolade_artifact_current_patron
		}
		if = {
			limit = {
				NOT = {
					artifact_owner = scope:miv_accolade_artifact_current_patron
				}
			}
			set_variable = {
				name = suppress_artifact_notifications
				value = yes
				days = 1
			}
		}
		miv_accolade_artifact_safekeep_item_effect = yes
	}
	else = {
		miv_accolade_artifact_unbind_item_effect = yes
		if = {
			limit = {
				NOT = {
					artifact_owner = scope:miv_accolade_artifact_taker
				}
			}
			set_variable = {
				name = suppress_artifact_notifications
				value = yes
				days = 1
			}
			unequip_artifact_from_owner = yes
			set_owner = {
				target = scope:miv_accolade_artifact_taker
			}
		}
		scope:miv_accolade_artifact_current_patron ?= {
			if = {
				limit = {
					NOT = {
						has_personal_artifact_claim = scope:miv_accolade_artifact_stolen
					}
					scope:miv_accolade_artifact_stolen = {
						NOT = {
							artifact_owner = scope:miv_accolade_artifact_current_patron
						}
					}
				}
				add_personal_artifact_claim = scope:miv_accolade_artifact_stolen
			}
			house ?= {
				save_scope_as = miv_accolade_artifact_claim_house
				if = {
					limit = {
						scope:miv_accolade_artifact_stolen = {
							NOT = {
								any_artifact_house_claimant = {
									this = scope:miv_accolade_artifact_claim_house
								}
							}
						}
					}
					add_house_artifact_claim = scope:miv_accolade_artifact_stolen
				}
			}
			send_interface_toast = {
				type = msg_artifact_lost
				title = miv_accolade_artifact_stolen_lost
				left_icon = scope:miv_accolade_artifact_slain_knight
				right_icon = scope:miv_accolade_artifact_stolen
				custom_tooltip = miv_accolade_artifact_stolen_lost_tt
			}
		}
		scope:miv_accolade_artifact_taker = {
			send_interface_toast = {
				type = event_artifact_good
				title = miv_accolade_artifact_stolen_won
				left_icon = scope:miv_accolade_artifact_slayer
				right_icon = scope:miv_accolade_artifact_stolen
				custom_tooltip = miv_accolade_artifact_stolen_won_tt
			}
		}
	}
}

miv_ai_artifacts_yearly_effect = {
	random_ruler = {
		limit = {
			miv_ai_artifacts_candidate_trigger = yes
		}
		weight = {
			base = 2
			modifier = {
				highest_held_title_tier = tier_kingdom
				add = 1
			}
			modifier = {
				highest_held_title_tier >= tier_empire
				add = 2
			}
		}
		add_character_flag = miv_ai_artifacts_received
		save_scope_value_as = {
			name = quality
			value = 40
		}
		save_scope_value_as = {
			name = wealth
			value = 40
		}
		if = {
			limit = {
				NOT = {
					any_character_artifact = {
						artifact_slot_type = primary_armament
					}
				}
			}
			create_artifact_weapon_effect = {
				OWNER = this
				CREATOR = this
				SET_WEAPON_TYPE = flag:no
			}
		}
		else = {
			create_artifact_armor_effect = {
				OWNER = this
				CREATOR = this
				SET_ARMOR_TYPE = flag:no
			}
		}
	}
}
