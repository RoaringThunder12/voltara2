voltara_ensure_ruins_realm = {
 if = { limit = { exists = global_var:voltara_ruins_keeper }
  title:e_voltara_ruins = {
   set_title_name = voltara_empty_ruins_name
   if = { limit = { NOT = { holder = global_var:voltara_ruins_keeper } }
    create_title_and_vassal_change = { type = granted save_scope_as = ruins_realm_change add_claim_on_loss = no }
    change_title_holder = { holder = global_var:voltara_ruins_keeper change = scope:ruins_realm_change }
    resolve_title_and_vassal_change = scope:ruins_realm_change
   }
  }
  global_var:voltara_ruins_keeper = {
   add_trait = voltara_ruins_keeper
   set_culture = culture:voltara_culture_44
   set_character_faith = faith:voltara_faith_0
   set_primary_title_to = title:e_voltara_ruins
   change_government = voltara_ruins_government
   if = { limit = { exists = liege }
    create_title_and_vassal_change = { type = independency save_scope_as = ruins_independence add_claim_on_loss = no }
    becomes_independent = { change = scope:ruins_independence }
    resolve_title_and_vassal_change = scope:ruins_independence
   }
  }
 }
}
