# One-time migration: later voluntary government reforms are not reversed.
voltara_apply_tribal_southron_update = {
 if = { limit = { NOT = { has_global_variable = voltara_tribal_southron_applied } }
  every_living_character = {
   limit = { is_landed = yes has_government = clan_government NOT = { culture = culture:voltara_culture_23 } }
   every_held_title = { limit = { tier = tier_county }
    title_province = {
     if = { limit = { OR = { has_holding_type = castle_holding has_holding_type = tribal_holding } } set_holding_type = tribal_holding }
    }
   }
   change_government = tribal_government
  }
  if = { limit = { exists = title:e_voltara_sixth_southron_invasion.holder }
   title:e_voltara_sixth_southron_invasion.holder = { save_scope_as = southron_nomad_leader }
   every_living_character = {
    limit = {
     is_landed = yes highest_held_title_tier >= tier_county
     OR = { culture = culture:voltara_southorion culture = culture:voltara_desert_southron }
     OR = { this = scope:southron_nomad_leader is_vassal_or_below_of = scope:southron_nomad_leader }
     NOT = { voltara_is_ruins_placeholder = yes }
    }
    every_held_title = { limit = { tier = tier_county }
     title_province = {
      if = { limit = { OR = { has_holding_type = castle_holding has_holding_type = tribal_holding has_holding_type = nomad_holding } } set_holding_type = nomad_holding }
     }
    }
    if = { limit = { NOT = { has_government = nomad_government } } change_government = nomad_government }
    if = { limit = { NOT = { has_trait = nomadic_philosophy } } add_trait = nomadic_philosophy }
   }
  }
  set_global_variable = { name = voltara_tribal_southron_applied value = yes }
 }
}
