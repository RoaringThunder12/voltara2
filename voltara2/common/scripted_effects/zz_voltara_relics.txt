voltara_enshrine_first_measure_effect = {
 if = { limit = { NOT = { has_global_variable = voltara_relic_first_measure_seeded } exists = title:c_voltara_4509.holder holy_site:voltara_first_measure = { holy_site_artifact_count < 3 } }
 title:c_voltara_4509.holder = { create_artifact = { name = voltara_relic_first_measure description = voltara_relic_first_measure_desc type = pedestal template = branch_general_template visuals = reliquary wealth = 30 quality = 60 history = { type = created_before_history } modifier = voltara_sanctuary_relic_modifier save_scope_as = voltara_new_relic } }
 scope:voltara_new_relic = { set_variable = { name = relic_religion value = religion:voltara_religion } set_holy_site_owner = { target = holy_site:voltara_first_measure generate_history = no } equip_artifact_to_owner = yes }
 set_global_variable = { name = voltara_relic_first_measure_seeded value = yes }
 }
}
