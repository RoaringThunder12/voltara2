agot_send_to_dragonpit_no_event = {
	$DRAGONPIT_COUNTY$ = {
		add_to_variable_list = {
			name = dragons_in_pit
			target = $DRAGON$
		}
	}
	$DRAGON$ = {
		add_character_flag = in_dragonpit
		add_character_modifier = agot_dragon_in_dragonpit
		set_variable = {
			name = pitted_dragon_location
			value = $DRAGONPIT_COUNTY$
		}
		remove_variable = travelling_from_pit
	}
}
agot_remove_from_dragonpit_no_event = {
	$DRAGONPIT_COUNTY$ = {
		remove_list_variable = {
			name = dragons_in_pit
			target = $DRAGON$
		}
		save_scope_as = dragon_pit_location
	}
	$DRAGON$ = {
		remove_character_flag = in_dragonpit
		remove_character_modifier = agot_dragon_in_dragonpit
		remove_variable = pitted_dragon_location
		if = {
			limit = { has_character_flag = dragon_put_in_different_pit }
			remove_character_flag = dragon_put_in_different_pit
		}
		save_scope_as = removed_dragon
	}
}
voltara_dragon_start_lifecycle_effect = {
 if = { limit = { NOT = { has_character_flag = voltara_lifecycle_started } }
  add_character_flag = voltara_lifecycle_started create_story = voltara_dragon_lifecycle
 }
}
voltara_dragon_unpit_effect = {
 if = { limit = { exists = var:pitted_dragon_location }
 save_scope_as = dragon
 agot_remove_from_dragonpit_no_event = { DRAGON = scope:dragon DRAGONPIT_COUNTY = var:pitted_dragon_location }
 }
 remove_character_flag = in_dragonpit remove_character_modifier = agot_dragon_in_dragonpit
}
voltara_dragon_pit_check_effect = {
 if = { limit = { exists = var:pitted_dragon_location }
  if = { limit = { var:pitted_dragon_location = { exists = holder any_county_province = { voltara_active_dragonpit = yes } } }
   set_location = var:pitted_dragon_location.title_province
   if = { limit = { NOT = { exists = var:current_rider } }
    save_scope_as = captive_dragon var:pitted_dragon_location.holder = { add_courtier = scope:captive_dragon }
   }
   else_if = { limit = { NOT = { var:pitted_dragon_location.holder = var:current_rider } }
    voltara_dragon_unpit_effect = yes set_location = var:current_rider.location
   }
  }
  else = { voltara_dragon_unpit_effect = yes }
 }

if = { limit = { exists = var:current_rider }
 if = { limit = { has_trait = dragon_glowing_scales }
  var:current_rider = { add_character_modifier = { modifier = voltara_glowing_dragon_prestige months = 2 } }
 }
 else = { var:current_rider = { remove_character_modifier = voltara_glowing_dragon_prestige } }
}
}
