voltara_register_dragon_effect = {
 add_trait = dragon add_character_flag = voltara_dragon
 set_variable = { name = dragon_size value = age } add_to_global_variable_list = { name = voltara_living_dragons target = this }
}
voltara_bond_dragon_effect = {
 if = { limit = { is_alive = yes NOT = { voltara_has_living_dragon = yes } scope:dragon = { voltara_free_dragon = yes } }
 save_scope_as = voltara_bond_owner
 set_variable = { name = current_dragon value = scope:dragon } add_trait = dragonrider
 scope:dragon = { set_variable = { name = current_rider value = scope:voltara_bond_owner } add_character_flag = owned_dragon }
 add_courtier = scope:dragon
 trigger_event = { id = voltara_dragons.90 days = 30 }
 }
}

 save_scope_as = voltara_hatcher
 create_character = { name = Voltara_Emberwing age = 0 gender_female_chance = 50 dynasty = none ethnicity = dragon
 culture = scope:voltara_hatcher.culture faith = scope:voltara_hatcher.faith location = scope:voltara_hatcher.capital_province
 random_traits = no trait = dragon trait = education_dragon_1 save_scope_as = dragon }
 scope:dragon = { voltara_register_dragon_effect = yes
 random_list = {
 1 = { change_first_name = Voltara_Emberwing } 1 = { change_first_name = Voltara_Dawnscale }
 1 = { change_first_name = Voltara_Ashwhisper } 1 = { change_first_name = Voltara_Nightflare }
 1 = { change_first_name = Voltara_Stormtalon } 1 = { change_first_name = Voltara_Sunfang }
 }
 if = { limit = { exists = scope:egg.var:dragon_mother } set_mother = scope:egg.var:dragon_mother }
 }

}
