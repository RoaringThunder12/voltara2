advanced_torture_reveal_prisoner_secret_effect = {
 scope:advanced_torture_prisoner = {
  if = {
   limit = {
    is_alive = yes

    any_secret = {
     NOT = {
      any_secret_knower = {
       this = root
      }
     }
    }
   }

   random_secret = {
    limit = {
     NOT = {
      any_secret_knower = {
       this = root
      }
     }
    }

    reveal_to = root
   }

   save_scope_as = advanced_torture_secret_bearer

   root = {
    advanced_torture_init_torture_variables_effect = yes

    change_variable = {
     name = advanced_torture_secret_reveals
     add = 1
    }

    advanced_torture_check_torture_nickname_effect = yes

    trigger_event = {
     id = advanced_torture.3001
    }
   }
  }
 }
}
