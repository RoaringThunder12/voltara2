 scope:defender = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }

		highest_held_title_tier = tier_empire

				top_liege = scope:defender.top_liege

 NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins }
 scope:attacker = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }
 scope:defender = { NOR = { has_trait = voltara_ruins_keeper has_government = voltara_ruins_government has_title = title:e_voltara_ruins } }

		highest_held_title_tier = tier_empire
        is_independent_ruler = yes
	}

	allowed_for_character_display_regardless = {
		prestige_level >= 4
		NOT = {
			has_character_flag = already_declared_great_empire_war_in_lifetime
		}
	}
	cost = {
		prestige = {
			value = 0
			if = {
				limit = {
					has_game_rule = full_restrictions
				}
				add = {
					value = 2000
					desc = CB_BASE_COST
				}
			}
			if = {
				limit = {
					has_game_rule = partial_restrictions
				}
				add = {
					value = 2000
					desc = CB_BASE_COST
				}
			}
			if = {
				limit = {
					scope:attacker = {
						is_ai = yes
					}
				}
				multiply = {
					value = 0
				}
			}
		}
	}
	on_declaration = {
		hidden_effect = {
			scope:attacker = { 
				set_relation_rival = scope:defender
			}
		}
		on_declared_war = yes
		scope:attacker = {
			if = {
				limit = {
					has_game_rule = full_restrictions
				}
				add_character_flag = already_declared_great_empire_war_in_lifetime
			}
			if = {
				limit = {
					has_game_rule = direct_only
					#NOT = { has_game_rule = call_the_banners_enabled }
				}
				every_vassal = {
					limit = {
						highest_held_title_tier >= tier_county
						max_military_strength > 500
						is_landed = yes
					}
					trigger_event = {
						id = great_empire_war.0005
					}
				}
			}

			hidden_effect = {
				add_character_modifier = {
					modifier = great_empire_war_modifier
					days = 3650
				}
			}
		}
		scope:defender = {
			if = {
				limit = {
					has_game_rule = full_restrictions
				}
				add_character_flag = already_declared_great_empire_war_in_lifetime
			}
			hidden_effect = {
				add_character_modifier = {
					modifier = great_empire_war_modifier

				}

				}

					limit = {

						is_landed = yes
					}
					trigger_event = {
						id = great_empire_war.0006
					}
				}
			}
		}
		scope:attacker = {
			trigger_event = great_empire_war.0003
		}
		scope:defender = {
			trigger_event = great_empire_war.0003
		}
	}
	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
					}
				}
				desc = great_empire_victory_desc_attacker
			}
			desc = great_empire_victory_desc
		}
	}
	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender = {
						is_local_player = yes
					}
				}
				desc = great_empire_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
					}
				}
				desc = great_empire_defeat_desc_attacker
			}
			desc = great_empire_defeat_desc
		}
	}
	on_victory = {
		scope:attacker = {
			trigger_event = great_empire_war.0001
		}
		scope:defender = {
			trigger_event = great_empire_war.0001
		}
		scope:attacker = {
			add_trait = victor_of_a_great_empire_war
			show_pow_release_message_effect = yes
		}
		scope:attacker.dynasty = {
			add_dynasty_prestige = 3000
		}
		add_from_contribution_attackers = {
			prestige = 8000
			opinion = {
				modifier = contributed_in_war
			}
		}
		scope:defender = {
			add_prestige = -4000
			#if = {

			every_vassal_or_below = {
				every_held_title = {
					if = {
						limit = {
							has_game_rule = full_shatter
							OR = {
								tier = tier_empire
								tier = tier_kingdom
								tier = tier_duchy
							}
						}
						holder = {
							destroy_title = prev
						}
					}
					else_if = {
						limit = {
							has_game_rule = partial_shatter
							OR = {
								tier = tier_empire
								tier = tier_kingdom
							}
						}
						holder = {
							destroy_title = prev
						}
					}
				}
				trigger_event = great_empire_war.0007
			}
			every_held_title = {
				if = {
					limit = {
						has_game_rule = full_shatter
						OR = {
							tier = tier_empire
							tier = tier_kingdom
							tier = tier_duchy
						}
					}
					holder = {
						destroy_title = prev
					}
				}
				else_if = {
					limit = {
						has_game_rule = partial_shatter
						OR = {
							tier = tier_empire
							tier = tier_kingdom
						}
					}
					holder = {
						destroy_title = prev
					}
				}
				else_if = {
					limit = {
						has_game_rule = minimal_shatter
						tier = tier_empire
						scope:defender.highest_held_title_tier = tier_empire
					}
					holder = {
						destroy_title = prev
					}
				}
				else_if = {
					limit = {
						has_game_rule = minimal_shatter
						tier = tier_kingdom
						scope:defender.highest_held_title_tier = tier_kingdom
					}
					holder = {
						destroy_title = prev
					}
				}
			}
			trigger_event = great_empire_war.0007
		}
	}

								tier = tier_empire
								tier = tier_kingdom
								tier = tier_duchy

							tier = tier_empire
							tier = tier_kingdom
							tier = tier_duchy
						}
					}
					holder = {
						destroy_title = prev
					}
				}
				else_if = {
					limit = {
						has_game_rule = partial_shatter
						OR = {
							tier = tier_empire
							tier = tier_kingdom
						}
					}
					holder = {
						destroy_title = prev
					}
				}
				else_if = {
					limit = {
						has_game_rule = minimal_shatter
						tier = tier_empire
