cow_generic_university_01_action = {
	priority = 100
	combine_into_one = no
	check_create_action = {
		every_held_title = {
			limit = {
				AND = {
					tier = tier_barony
					title_province = { has_building = cow_generic_university_01 }
				}
				trigger_if = {
					limit = {
						# Easy Dev Limit
						has_game_rule = easy_dev_requirement
					}
					county = {
						development_level >= 50.00
					}
				}
				trigger_if = {
					limit = {
						# Normal Dev Limit
						has_game_rule = normal_dev_requirement
					}
					county = {
						development_level >= 75.00
					}			
				}

			}
			try_create_important_action = {
				important_action_type = cow_generic_university_01_action
				actor = root
				landed_title = this
			}
		}
	}
	effect = {
		scope:landed_title.title_province = {
			open_view_data = {
				view = holding_view
			}
		}
	}
}
