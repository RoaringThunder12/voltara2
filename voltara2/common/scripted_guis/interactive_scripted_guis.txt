miv_bannermen_widget_init = {
	scope = character

	effect = {
		if = {
			limit = {
				is_ai = no
				has_game_rule = bannermen_mode_enabled
				interactive_bannermen_foreign_enabled_trigger = yes
				NOT = { miv_gui_banners_cache_valid_trigger = yes }
			}
			save_scope_as = liege
			interactive_bannermen_preview_effect = yes
			set_variable = {
				name = miv_widget_headline_eligible_strength
				value = interactive_bannermen_eligible_strength
			}
			set_variable = {
				name = miv_widget_headline_eligible_count
				value = interactive_bannermen_eligible_count
			}
			set_variable = {
				name = miv_widget_headline_answer_strength
				value = miv_widget_preview_answering_strength
			}
			set_variable = {
				name = miv_widget_headline_answer_count
				value = interactive_bannermen_answering_count
			}
			set_variable = {
				name = miv_banners_view_answering_count
				value = miv_banners_summary_answering_count
			}
			set_variable = {
				name = miv_banners_view_answering_strength
				value = miv_banners_summary_answering_strength
			}
			set_variable = {
				name = miv_banners_view_withheld_count
				value = miv_banners_summary_withheld_count
			}
			set_variable = {
				name = miv_banners_view_withheld_strength
				value = miv_banners_summary_withheld_strength
			}
			set_variable = {
				name = miv_banners_view_betraying_count
				value = miv_banners_summary_betraying_count
			}
			set_variable = {
				name = miv_banners_view_betraying_strength
				value = miv_banners_summary_betraying_strength
			}
			set_variable = {
				name = miv_banners_view_distant_count
				value = miv_banners_summary_distant_count
			}
			set_variable = {
				name = miv_banners_view_distant_strength
				value = miv_banners_summary_distant_strength
			}
			interactive_bannermen_widget_fill_effect = yes
			add_character_flag = { flag = miv_gui_banners_ready days = 3 }
		}
	}
}

miv_friendly_war_participant = {
	scope = character
	is_shown = {
		has_game_rule = miv_vassal_map_gfx_enabled
		exists = scope:player

			is_at_war = yes
		}
		is_at_war = yes
		NOT = {
			this = scope:player
		}
		any_war_ally = {

		}
		NOT = {
			is_at_war_with = scope:player
		}
	}
}

miv_gui_character_capture = {
	scope = character
	is_shown = {
		miv_gui_character_capture_needed_trigger = yes
	}
	is_valid = {
		miv_gui_character_capture_needed_trigger = yes
	}
	effect = {
		if = {
			limit = {
				miv_gui_character_capture_needed_trigger = yes
			}
			set_variable = {
				name = miv_gui_strength_eligible_count
				value = interactive_bannermen_eligible_count
			}
			set_variable = {
				name = miv_gui_strength_eligible_strength
				value = interactive_bannermen_eligible_strength
			}
			set_variable = {
				name = miv_gui_strength_not_answering_count
				value = miv_bannermen_not_answering_count
			}
			save_temporary_scope_value_as = {
				name = miv_gui_not_answering_strength
				value = miv_bannermen_not_answering_strength
			}
			set_variable = {
				name = miv_gui_strength_answering_count
				value = {
					value = var:miv_gui_strength_eligible_count
					subtract = var:miv_gui_strength_not_answering_count
				}
			}
			set_variable = {
				name = miv_gui_strength_answering_strength
				value = {
					value = var:miv_gui_strength_eligible_strength
					subtract = scope:miv_gui_not_answering_strength
				}
			}
			add_character_flag = { flag = miv_gui_character_ready days = 3 }
		}
	}
}
