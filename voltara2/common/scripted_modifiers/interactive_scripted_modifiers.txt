		desc = MIV_FACTION_POWER_THRESHOLD_RULE
		miv_faction_threshold_applies_trigger = yes
		add = {
			value = miv_faction_threshold_level
			add = -80
		}
	}
	modifier = {
		add = {
			value = 0
			if = {
				limit = {
					faction_is_at_war = no
					faction_target = {
						any_character_war = {
							has_variable = miv_civil_war
							primary_defender = prev
						}
					}
				}
				add = 1000
			}
			if = {
				limit = {
					faction_target = {
						any_targeting_faction = {

					}

				add = -20

				limit = {
