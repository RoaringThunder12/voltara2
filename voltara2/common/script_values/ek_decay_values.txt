# Calculate the monthly change for an individuals decay level
monthly_decay_increase = {
	value = 0.01
	add = {
        fixed_range = {
            min = small_decay_increase
            max = medium_decay_increase
        }
    }
    multiply = arcana_decay_mult
}

arcana_decay_mult = {
    value = arcana
    subtract = 10
    min = 1
    divide = 25
    min = 0.1
    max = 1.0
}
