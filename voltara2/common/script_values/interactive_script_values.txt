interactive_bannermen_muster_score = {
	value = 0
	if = {
		limit = {
			interactive_bannermen_rebellion_trigger = {
				LIEGE = scope:liege
			}
		}
		add = {
			value = -50
			desc = interactive_bannermen_desc_state_rebellion
		}
	}
	if = {
		limit = {
			is_imprisoned = yes
		}
		add = {
			value = -25
			desc = interactive_bannermen_desc_state_prison
		}
	}
	if = {
		limit = {
			is_incapable = yes
		}
		add = {
			value = -25
			desc = interactive_bannermen_desc_state_incapable
		}
	}
	if = {
		limit = {
			is_adult = no
		}
		add = {
			value = -25
			desc = interactive_bannermen_desc_state_child
		}
	}
	if = {
		limit = {
			any_character_war = {
				primary_defender = scope:vassal
				primary_attacker = {
					interactive_bannermen_is_rebel_trigger = {
						LIEGE = scope:vassal
					}
				}
